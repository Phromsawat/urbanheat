import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

# กำหนดปีที่ต้องการ (8 ปี: 2018 - 2025)
years = [2018, 2019, 2020, 2021, 2022, 2023, 2024, 2025]
gas_types = ['CH4', 'CO', 'NO2', 'O3', 'SO2', 'LST']

# สร้างตารางสำหรับวาด 8 กราฟ (4 แถว 2 คอลัมน์)
fig, axes = plt.subplots(4, 2, figsize=(15, 22))
axes = axes.flatten()

for i, year in enumerate(years):
    # สร้าง dictionary เพื่อเก็บข้อมูลของปีนั้นๆ แบบจับคู่แก๊สให้ตรงปีจริง
    year_data = {}

    for gas in gas_types:
        expected_col = f"{gas}_{year}"
        # ตรวจสอบว่ามีคอลัมน์นี้อยู่ใน df_clean ไหม (ป้องกันปัญหา SO2 หรือตัวอื่นสลับที่)
        if expected_col in df_clean.columns:
            year_data[gas] = df_clean[expected_col]

    # แปลงข้อมูลของปีนั้นเป็น DataFrame ชั่วคราว
    df_year = pd.DataFrame(year_data)

    # คำนวณค่าความสัมพันธ์ (Correlation) ของปีนั้นๆ
    corr_matrix = df_year.corr()

    # วาด Heatmap
    sns.heatmap(
        corr_matrix,
        annot=True,
        cmap='coolwarm',
        vmin=-1, vmax=1,
        fmt=".2f",
        linewidths=0.5,
        ax=axes[i],
        cbar=(i % 2 != 0) # เปิดแถบสีเฉพาะกราฟฝั่งขวาเพื่อความสวยงาม
    )

    axes[i].set_title(f'Gas vs LST Correlation - Year {year}', fontsize=13, fontweight='bold', pad=10)

plt.tight_layout()
output_image_name = 'Chonburi_Gas_LST_Correlation_Heatmaps.png'
plt.savefig(output_image_name, dpi=300, bbox_inches='tight')
print(f"[สำเร็จ] บันทึกไฟล์ภาพความละเอียดสูงเรียบร้อยในชื่อ: {output_image_name}")

gases = ['CH4', 'CO', 'NO2', 'O3', 'SO2']
df_grouped = df_long.groupby('LULC')[gases].mean().reset_index()

# ปรับสเกลข้อมูลให้อยู่ในสเกล 0-1
scaler = MinMaxScaler()
df_scaled_metrics = pd.DataFrame(scaler.fit_transform(df_grouped[gases]), columns=gases)
df_scaled_metrics['LULC'] = df_grouped['LULC']

# เปลี่ยนโครงสร้างเพื่อใช้วาดกราฟแท่งเปรียบเทียบ
df_melted = pd.melt(df_scaled_metrics, id_vars=['LULC'], value_vars=gases,
                    var_name='Gas_Type', value_name='Scaled_Concentration')

# =========================================================
plt.figure(figsize=(10, 7))  # กำหนดขนาดรูปแยกต่างหาก

plt.title('Heatmap: Relationship between LULC and Gas Concentrations (Scaled)', fontsize=13, pad=15)
plt.xlabel('Gas Type')
plt.ylabel('Land Use / Land Cover (LULC)')
plt.tight_layout()
